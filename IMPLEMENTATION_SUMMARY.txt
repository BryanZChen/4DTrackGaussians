# KEY FORMULAS
# ============================================================================

FORMULAS = """
1. OPTICAL FLOW LOSS
   ───────────────────

   L_flow = (1/N) Σᵢ αᵢ ||Δp_2D^(i) - F_gt(u_t^(i))||_1
   
   where:
   - Δp_2D^(i) = Proj(μ_{t+1}^(i)) - Proj(μ_t^(i))  [predicted 2D motion]
   - F_gt(u_t^(i))  [ground-truth flow sampled at 2D position]
   - αᵢ = opacity (weight transparent regions down)


2. RIGIDITY LOSS
   ───────────────

   L_rigid = (1/M) Σⱼ ||Δd^(j)||²_2
   
   where:
   - Δd^(j) = d(xⱼ) - d(xⱼ + ε·uⱼ)  [deformation difference]
   - d(x) = deformation vector
   - ε·uⱼ = small random perturbation


3. COARSE-TO-FINE SCHEDULING
   ──────────────────────────

   During warmup (t < t_warmup):
   - Resolution: 1/s (s=downsample_factor, e.g., 4)
   - Losses: L1 only
   
   During transition (t_warmup ≤ t < t_warmup + Δt_trans):
   - Resolution: (1/s) + (1 - 1/s) × (t - t_warmup) / Δt_trans
   - Flow weight: λ_0 + (λ_1 - λ_0) × progress
   
   During fine (t ≥ t_warmup + Δt_trans):
   - Resolution: 1 (full)
   - Losses: L1 + λ_flow × L_flow + λ_rigid × L_rigid


4. 3D TO 2D PROJECTION
   ────────────────────

   View transform:  p_view = V × p_world   [4×4 × 4×1]
   Projection:      p_proj = P × p_view    [4×4 × 4×1]
   Perspective div: p_ndc = p_proj.xy / p_proj.w
   Result:          p_2d ∈ [-1, 1] (normalized device coordinates)
"""

HYPERPARAMETERS = """
CRITICAL PARAMETERS (tune first)
═════════════════════════════════════════════════════════════════════════

lambda_flow (flow loss weight)
  ├─ Range: [0.05, 1.0]
  ├─ Default: 0.2
  ├─ Fast motion: ↑ (0.3-0.5)
  ├─ Subtle motion: ↓ (0.05-0.1)
  └─ Too high: → training instability, NaN losses
            → reduce to 0.05 and ramp gradually

lambda_rigidity (smoothness regularization)
  ├─ Range: [0.0, 0.1]
  ├─ Default: 0.01
  ├─ Fast motion: ↑ (0.02-0.05)
  ├─ Subtle motion: ↓ (0.005-0.01)
  └─ Too high: → over-smoothing, loss of details


SCHEDULING PARAMETERS (tune second)
═════════════════════════════════════════════════════════════════════════

coarse_warmup_iters (length of low-resolution warmup)
  ├─ Range: [1000, 5000]
  ├─ Default: 3000
  ├─ Fast motion: ↑ (5000)     [longer warmup helps]
  ├─ Simple scenes: ↓ (1000)   [less needed]
  └─ Effect: longer = more coarse-level learning

downsample_factor (resolution reduction during warmup)
  ├─ Range: [2, 8]
  ├─ Default: 4 (4x4 = 16x memory savings)
  ├─ Memory constrained: ↑ (8)
  ├─ Fast hardware: ↓ (2)
  └─ Typical: 4

lambda_flow_start, lambda_flow_end (flow weight scheduling)
  ├─ Typical: 0.1 → 0.5
  ├─ Aggressive: 0.2 → 1.0
  ├─ Conservative: 0.05 → 0.2
  └─ Effect: higher end = stronger fine-stage motion constraint


SAMPLING PARAMETERS (tune third, if needed)
═════════════════════════════════════════════════════════════════════════

num_samples_per_gaussian (for rigidity loss)
  ├─ Range: [1, 10]
  ├─ Default: 3
  ├─ Memory OK: 5
  ├─ Memory tight: 1
  └─ Effect: more samples = more robust but slower

neighborhood_radius (perturbation magnitude for rigidity)
  ├─ Range: [0.05, 1.0]
  ├─ Default: 0.5
  ├─ Smaller objects: ↓ (0.1-0.2)
  ├─ Larger objects: ↑ (1.0)
  └─ Effect: local neighborhood size for smoothness

perturbation (magnitude for rigidity loss)
  ├─ Range: [0.05, 0.5]
  ├─ Default: 0.1
  ├─ Coarse models: ↑ (0.2)
  ├─ Fine models: ↓ (0.05)
  └─ Effect: how much we perturb neighbors
"""

TROUBLESHOOTING = """
Issue: Flow loss stays at 0.0
─────────────────────────────────────────────────────────────────────────
Cause:   camera.flow_map is all zeros or None
Fix:     1. Verify flow files exist: ls flow/flow_*.npy
         2. Check filenames match pattern: flow_000000.npy, flow_000001.npy
         3. Print debug info: print(camera.flow_map.shape, camera.flow_map.sum())
         4. Ensure flow_dir is passed to FourDGSdataset


Issue: Flow loss becomes NaN
─────────────────────────────────────────────────────────────────────────
Cause:   Large flow values or projection errors
Fix:     1. Reduce lambda_flow to 0.05 (conservative start)
         2. Check optical flow range: min/max should be < 50 pixels
         3. Verify projection matrix is correct
         4. Print flow stats: print(flow.min(), flow.max(), flow.mean())
         5. Check camera intrinsics are valid


Issue: Total loss increases at iteration 3000
─────────────────────────────────────────────────────────────────────────
Cause:   Normal at resolution transition
Fix:     1. This is expected behavior (transition point)
         2. Wait 200-500 iterations for recovery
         3. If still increasing, check:
            - Focal length adjustment is correct
            - Image downsampling is working
            - Camera intrinsics are being adjusted


Issue: GPU out of memory
─────────────────────────────────────────────────────────────────────────
Fix:     1. Reduce batch_size to 1
         2. Reduce num_samples_per_gaussian to 1
         3. Disable rigidity loss (lambda_rigidity = 0)
         4. Increase downsample_factor to 8
         5. Monitor with: nvidia-smi


Issue: No visual improvement despite flow loss
─────────────────────────────────────────────────────────────────────────
Fix:     1. Verify loss_weights['flow_enabled'] is True after warmup
         2. Check camera.time values are reasonable
         3. Ensure time_next is correctly set (time + 1.0)
         4. Try increasing lambda_flow gradually
         5. Print: print(loss_weights, resolution_scale)
"""
